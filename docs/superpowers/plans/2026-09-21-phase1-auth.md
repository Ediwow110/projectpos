# Phase 1 Authentication and Business Setup Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement Phase 1 per `docs/superpowers/specs/2026-09-21-phase1-auth-design.md` so an owner can register, create a business, pass the onboarding wizard, and reach an empty dashboard, with cross-tenant access blocked throughout.

**Architecture:** NestJS modular monolith with three modules — AuthModule (register/login/logout/reset/sessions/re-auth), TenantModule (business, membership, role seed, active-business switch), BusinessModule (settings, onboarding state). Pure domain logic lives in framework-free functions under `apps/api/src/` and is unit-tested directly; NestJS controllers are thin adapters tested via `supertest`. Cross-module calls use NestJS services only, never direct table access.

**Tech Stack:** TypeScript strict, Node 24, pnpm 9.15.9, Turborepo, NestJS 11 (REST, `/api/v1`), PostgreSQL 16, Prisma, argon2id (`argon2` npm), Zod 3.23.8, Vitest 2.1.8, supertest, Next.js + React + Tailwind + TanStack Query for web.

**Spec:** `docs/superpowers/specs/2026-09-21-phase1-auth-design.md`. PLAN.md (`G:\Vscode\PLAN.md`) is authoritative on conflicts, the spec second, this plan third.

## Global Constraints

- TypeScript strict end-to-end, integer minor units for money, no float totals (unchanged from Phase 0).
- Tenant derived server-side from authenticated membership only, never from client tenantId claim (FR-TENANT-01).
- Every business-owned table includes businessId, every read and write scoped via `withScope`.
- API versioned `/api/v1`, error envelope `{ code, message, correlationId, fields? }`, ISO 8601 UTC on wire, `X-Request-Id` correlation.
- Audit log append-only; password hashes and reset tokens redacted, never stored in snapshots.
- Passwords: argon2id, min 12 / max 128 chars; sessions 30 days; re-auth window 15 minutes; login rate limit 5/min/IP; reset resend cooldown 60 seconds/IP; reset tokens 15-minute single-use.
- Cookie `pos_session`: httpOnly, Secure in production, SameSite=Lax, Path=/; mutations need `X-CSRF-Token` double-submit header.
- Secrets outside source control; CI runs typecheck, lint, unit plus integration tests, migration check, build.
- No catalog, inventory ledger, POS UI, payments, shifts, reports, branches, offline, billing, integrations, or AI.
- Every task follows TDD red-green-refactor, minimal code, frequent commits.

```ts
import * as argon2 from "argon2";

export function checkPasswordPolicy(plain: string): void {
  if (plain.length < 12 || plain.length > 128) throw new Error("AUTH_WEAK_PASSWORD");
}

export async function hashPassword(plain: string): Promise<string> {
  checkPasswordPolicy(plain);
  return argon2.hash(plain, {
    type: argon2.argon2id,
    memoryCost: 65536,
    timeCost: 3,
    parallelism: 4,
  });
}

export async function verifyPassword(hash: string, plain: string): Promise<boolean> {
  try {
    return await argon2.verify(hash, plain);
  } catch {
    return false;
  }
}
```

Create `apps/api/src/auth/sessions.ts`:

```ts
import { createHash, randomBytes } from "node:crypto";

export const SESSION_TTL_MS = 30 * 24 * 3600 * 1000;
export const REAUTH_WINDOW_MS = 15 * 60 * 1000;

export interface SessionBuild {
  userId: string;
  businessId: string;
  ip: string | null;
  userAgent: string | null;
}

export function createSessionToken(): string {
  return randomBytes(32).toString("base64url");
}

export function hashToken(token: string): string {
  return createHash("sha256").update(token).digest("hex");
}

export function buildSessionRecord(input: SessionBuild, now: Date = new Date()): {
  token: string;
  tokenHash: string;
  userId: string;
  businessId: string;
  expiresAt: Date;
  createdAt: Date;
  ip: string | null;
  userAgent: string | null;
} {
  const token = createSessionToken();
  return {
    token,
    tokenHash: hashToken(token),
    userId: input.userId,
    businessId: input.businessId,
    expiresAt: new Date(now.getTime() + SESSION_TTL_MS),
    createdAt: now,
    ip: input.ip,
    userAgent: input.userAgent,
  };
}

export function isReAuthFresh(authenticatedAt: Date, now: Date = new Date()): boolean {
  return now.getTime() - authenticatedAt.getTime() <= REAUTH_WINDOW_MS;
}
```

Create `apps/api/src/auth/reset-tokens.ts`:

```ts
import { createSessionToken, hashToken } from "./sessions";

export const RESET_TTL_MS = 15 * 60_000;

export interface ResetRecord {
  userId: string;
  tokenHash: string;
  expiresAt: Date;
  usedAt: Date | null;
  createdAt: Date;
  requestId: string;
  ip: string | null;
}

export function createResetToken(input: {
  userId: string;
  requestId: string;
  ip: string | null;
  now?: Date;
}): { token: string } & ResetRecord {
  const now = input.now ?? new Date();
  const token = createSessionToken();
  return {
    token,
    userId: input.userId,
    tokenHash: hashToken(token),
    expiresAt: new Date(now.getTime() + RESET_TTL_MS),
    usedAt: null,
    createdAt: now,
    requestId: input.requestId,
    ip: input.ip,
  };
}

export function consumeResetToken(record: ResetRecord, now: Date = new Date()): void {
  if (record.usedAt !== null) throw new Error("AUTH_TOKEN_USED");
  if (record.expiresAt.getTime() <= now.getTime()) throw new Error("AUTH_TOKEN_EXPIRED");
}
```

Create `apps/api/src/auth/csrf.ts`:

```ts
import { createHash, randomBytes, timingSafeEqual } from "node:crypto";

export function issueCsrfToken(): string {
  return randomBytes(32).toString("base64url");
}

export function verifyCsrfToken(expected: string, presented: string | undefined): void {
  if (!presented) throw new Error("CSRF_REQUIRED");
  const a = Buffer.from(createHash("sha256").update(expected).digest("hex"));
  const b = Buffer.from(createHash("sha256").update(presented).digest("hex"));
  if (a.length !== b.length || !timingSafeEqual(a, b)) throw new Error("CSRF_MISMATCH");
}
```

```ts
import { SENSITIVE_ACTIONS } from "@pos/auth";
import { hashPassword } from "../auth/password";

export const ALL_PERMISSIONS: readonly string[] = SENSITIVE_ACTIONS as unknown as readonly string[];

export const ROLE_SEED: Record<"owner" | "manager" | "cashier" | "inventory", string[]> = {
  owner: [...ALL_PERMISSIONS],
  manager: ALL_PERMISSIONS.filter((p) => p !== "user.manage" && p !== "settings.write"),
  cashier: ALL_PERMISSIONS.filter(
    (p) =>
      p !== "price.edit" &&
      p !== "cost.view" &&
      p !== "sale.refund" &&
      p !== "sale.void" &&
      p !== "settings.write" &&
      p !== "user.manage" &&
      p !== "export.run",
  ),
  inventory: ALL_PERMISSIONS.filter(
    (p) => p !== "sale.refund" && p !== "user.manage" && p !== "settings.write",
  ),
};

export interface RegistrationInput {
  ownerName: string;
  email: string;
  password: string;
  businessName: string;
}

export interface RegistrationSeed {
  user: { name: string; email: string; passwordHash: string };
  business: { name: string; currency: string };
  ownerRoleName: string;
  roles: Record<string, string[]>;
  allPermissions: readonly string[];
  defaultSettings: Array<{ key: string; value: unknown }>;
  onboarding: { currentStep: string; steps: Record<string, "done" | "skipped" | "pending"> };
}

export async function buildRegistrationSeed(input: RegistrationInput): Promise<RegistrationSeed> {
  const passwordHash = await hashPassword(input.password);
  return {
    user: { name: input.ownerName, email: input.email, passwordHash },
    business: { name: input.businessName, currency: "PHP" },
    ownerRoleName: "owner",
    roles: {
      owner: ROLE_SEED.owner,
      manager: ROLE_SEED.manager,
      cashier: ROLE_SEED.cashier,
      inventory: ROLE_SEED.inventory,
    },
    allPermissions: ALL_PERMISSIONS,
    defaultSettings: [{ key: "currency", value: "PHP" }],
    onboarding: {
      currentStep: "taxCurrency",
      steps: {
        businessDetails: "done",
        taxCurrency: "pending",
        receiptSetup: "pending",
        products: "pending",
        employees: "pending",
      },
    },
  };
}

export function resolveActiveBusiness(membershipBusinessIds: string[], requestedId: string): string {
  if (!membershipBusinessIds.includes(requestedId)) throw new Error("TENANT_MISMATCH");
  return requestedId;
}
```

```ts
export const ONBOARDING_ORDER = [
  "businessDetails",
  "taxCurrency",
  "receiptSetup",
  "products",
  "employees",
] as const;

export type OnboardingStep = (typeof ONBOARDING_ORDER)[number];
export type StepState = "done" | "skipped" | "pending";

export const REQUIRED_STEPS: readonly OnboardingStep[] = [
  "businessDetails",
  "taxCurrency",
  "receiptSetup",
];

export const SKIPPABLE_STEPS: readonly OnboardingStep[] = ["products", "employees"];

export interface OnboardingState {
  currentStep: string;
  steps: Record<OnboardingStep, StepState>;
}

export function defaultOnboardingState(): OnboardingState {
  return {
    currentStep: "taxCurrency",
    steps: {
      businessDetails: "done",
      taxCurrency: "pending",
      receiptSetup: "pending",
      products: "pending",
      employees: "pending",
    },
  };
}

export function applyOnboardingPatch(
  state: OnboardingState,
  patch: { step: OnboardingStep; state: "done" | "skipped" },
): OnboardingState {
  if (patch.state === "skipped" && !SKIPPABLE_STEPS.includes(patch.step)) {
    throw new Error("ONBOARDING_REQUIRED_STEP");
  }
  const steps = { ...state.steps, [patch.step]: patch.state };
  const next = ONBOARDING_ORDER.find((s) => steps[s] === "pending") ?? "complete";
  return { currentStep: next, steps };
}

export function isOnboardingComplete(state: OnboardingState): boolean {
  return REQUIRED_STEPS.every((s) => state.steps[s] === "done");
}

export function mergeSettings(
  current: Record<string, unknown>,
  patch: Record<string, unknown>,
): Record<string, unknown> {
  return { ...current, ...patch };
}
```

```ts
export class ApiError extends Error {
  code: string;
  correlationId: string;
  status: number;
  constructor(code: string, message: string, correlationId: string, status: number) {
    super(message);
    this.code = code;
    this.correlationId = correlationId;
    this.status = status;
  }
}

export async function apiFetch<T>(
  url: string,
  opts: { csrfToken?: string; init?: RequestInit } = {},
): Promise<T> {
  const res = await fetch(url, {
    ...opts.init,
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      ...(opts.csrfToken ? { "X-CSRF-Token": opts.csrfToken } : {}),
      ...(opts.init?.headers ?? {}),
    },
  });
  const body = (await res.json()) as T & { code?: string; message?: string; correlationId?: string };
  if (!res.ok) {
    throw new ApiError(
      body.code ?? "UNKNOWN",
      body.message ?? "Request failed",
      body.correlationId ?? "",
      res.status,
    );
  }
  return body as T;
}
```
